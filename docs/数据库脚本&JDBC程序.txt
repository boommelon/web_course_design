package bean;

public class Customer {
	private String id, name, phone;
    public Customer() {
    	
    }
    
	public Customer(String id, String name, String phone) {
		super();
		this.id = id;
		this.name = name;
		this.phone = phone;
	}

	public String getId() {
		return id;
	}

	public void setId(String id) {
		this.id = id;
	}

	public String getName() {
		return name;
	}

	public void setName(String name) {
		this.name = name;
	}

	public String getPhone() {
		return phone;
	}

	public void setPhone(String phone) {
		this.phone = phone;
	}
}

package dao;
import java.util.List;
import java.sql.*;
import java.util.ArrayList;
import bean.Customer;
public class CustomerDao {
  private String driver="com.mysql.jdbc.Driver";
  private String url="jdbc:mysql://127.0.0.1:3306/support",dbuser="root",dbpwd="1234";
  public List<Customer> queryAllCustomers(){
	  List<Customer> cusList=new ArrayList<Customer>();
	  try {
		  Class.forName(driver);
		  Connection conn=DriverManager.getConnection(url,dbuser,dbpwd);
		  Statement cmd=conn.createStatement();
		  ResultSet rs=cmd.executeQuery("select * from customers");
		  while(rs.next()) {
			  String id=rs.getString("customerID");
			  String name=rs.getString("name");
			  String phone=rs.getString("phone");
			  Customer cus=new Customer(id,name,phone);
			  cusList.add(cus);
		  }
		  conn.close();
	  }catch(Exception ex) {
		  ex.printStackTrace();
	  }
	  return cusList;
  }
  public static void main(String[] args) {
	  CustomerDao dao=new CustomerDao();
	  System.out.println(dao.queryAllCustomers().size());
  }
}
